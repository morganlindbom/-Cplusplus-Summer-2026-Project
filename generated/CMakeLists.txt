# CMakeLists.txt
cmake_minimum_required(VERSION 3.13)
include(pico_sdk_import.cmake)
project(PICO2W LANGUAGES C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
add_executable(PICO2W main.cpp)
target_link_libraries(PICO2W pico_stdlib hardware_gpio hardware_pwm hardware_uart hardware_spi hardware_i2c hardware_adc hardware_pio pico_cyw43_arch_lwip_threadsafe_background pico_lwip_nosys)
target_include_directories(PICO2W PRIVATE ${CMAKE_CURRENT_LIST_DIR})
pico_enable_stdio_usb(PICO2W 1)
pico_enable_stdio_uart(PICO2W 0)
pico_add_extra_outputs(PICO2W)
